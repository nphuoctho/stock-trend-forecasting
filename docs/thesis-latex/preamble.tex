% ============================================================================
% preamble.tex - Cấu hình hình thức trình bày theo Phụ lục 2 (quy định trường)
% Font Times New Roman 13pt, giãn dòng 1.5, lề T3/D3.5/L3.5/P2 cm.
% Biên dịch bằng XeLaTeX (hỗ trợ font hệ thống + tiếng Việt Unicode).
% ============================================================================

\usepackage{fontspec}          % dùng font hệ thống (XeLaTeX/LuaLaTeX)
\usepackage{polyglossia}       % đa ngôn ngữ, hỗ trợ tiếng Việt
\setmainlanguage{vietnamese}
\setotherlanguage{english}

% --- Font: Times New Roman (fallback Liberation Serif nếu máy không có TNR) ---
% Nếu máy có "Times New Roman", đổi dòng dưới thành \setmainfont{Times New Roman}.
\IfFontExistsTF{Times New Roman}{
  \setmainfont{Times New Roman}
}{
  \setmainfont{Liberation Serif}  % thay thế metric-compatible với Times New Roman
}

% --- Cỡ chữ 13pt: article mặc định tối đa 12pt, nên scale bằng anyfontsize ---
\usepackage{anyfontsize}
\usepackage{setspace}

% --- Lề trang: trên 3cm, dưới 3.5cm, trái 3.5cm, phải 2cm ---
\usepackage[
  top=3cm, bottom=3.5cm, left=3.5cm, right=2cm,
  a4paper
]{geometry}

% --- Giãn dòng 1.5 ---
\onehalfspacing

% --- Đánh số mục tới 4 cấp (Chương / x.y / x.y.z / x.y.z.t) ---
\usepackage{titlesec}
\setcounter{secnumdepth}{4}
\setcounter{tocdepth}{3}

% Chương: bold, cỡ ~14pt; mục con: bold, cỡ 13pt (theo quy định).
\titleformat{\chapter}[block]
  {\normalfont\fontsize{14}{18}\bfseries}
  {Chương \thechapter.}{0.6em}{}
\titlespacing*{\chapter}{0pt}{0pt}{12pt}

\titleformat{\section}
  {\normalfont\fontsize{13}{18}\bfseries}{\thesection.}{0.5em}{}
\titleformat{\subsection}
  {\normalfont\fontsize{13}{18}\bfseries}{\thesubsection.}{0.5em}{}
\titleformat{\subsubsection}
  {\normalfont\fontsize{13}{18}\bfseries}{\thesubsubsection.}{0.5em}{}

% --- Bảng, hình, caption đánh số theo chương (Hình 1.1, Bảng 2.1) ---
\usepackage{graphicx}
\usepackage{float}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage[labelsep=colon]{caption}
\captionsetup{font=small, labelfont=bf}
\usepackage{chngcntr}
\counterwithin{figure}{chapter}
\counterwithin{table}{chapter}

% --- Đánh số trang: giữa bên dưới (quy định) ---
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}

% --- Trích dẫn IEEE (số [1], [2]) ---
\usepackage[backend=biber, style=ieee, sorting=nyt]{biblatex}

% --- Tiện ích ---
\usepackage{hyperref}
\hypersetup{colorlinks=false, hidelinks, pdfborder={0 0 0}}
\usepackage{xurl}
\usepackage{enumitem}
\usepackage{amsmath, amssymb}
